import datetime

class AuthSystem:
    def __init__(self):
        # Cơ sở dữ liệu người dùng
        self.users_db = {}

    def register(self, username, password):
        if username in self.users_db:
            return False, "❌ Tài khoản này đã tồn tại!"
        self.users_db[username] = password
        return True, "✅ Đăng ký thành công! Hãy đăng nhập."

    def login(self, username, password, storage_history):
        # 1. Kiểm tra tài khoản tồn tại
        if username not in self.users_db:
            return False, "❌ Tài khoản không tồn tại!"
            
        # 2. Kiểm tra mật khẩu
        if self.users_db[username] != password:
            return False, "❌ Mật khẩu không chính xác!"

        # 3. Đúng thông tin -> Ghi nhận Lịch sử vào storage
        log_entry = {
            "user": username,
            "timestamp": str(datetime.datetime.now()),
            "status": "SUCCESS"
        }
        storage_history.append(log_entry)
        
        return True, f"🎉 Đăng nhập thành công! Chào {username}."
